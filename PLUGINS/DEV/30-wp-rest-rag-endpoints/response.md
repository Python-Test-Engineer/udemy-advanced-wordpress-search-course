SELECT                post_id,                post_title,                post_content,                categories,                tags,                MATCH(post_title, post_content)                AGAINST ('FOAM' IN NATURAL LANGUAGE MODE) as relevance_score            FROM wp_posts_rag            WHERE MATCH(post_title, post_content)                AGAINST ('FOAM' IN NATURAL LANGUAGE MODE)            ORDER BY relevance_score DESC            LIMIT 3
